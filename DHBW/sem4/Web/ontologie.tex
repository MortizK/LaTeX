\subsection{Bewertungskriterien}

\begin{itemize}
    \item Umfang der Ontologie
    \begin{itemize}
        \item bloße Anzahl von Entitäten kein Kriterium
        \item Klassen anspruchsvoller als Individuen
        \item Komplexe Zusammenhänge anspruchsvoller als baumförmige Hierarchien
    \end{itemize}
    \item Korrektheit und Adäquatheit der Modellierung
    \begin{itemize}
        \item Klassenhierarchie korrekt?
        \item Zuordnung Klasse/Individuum sinnvoll und konsistent?
        \item Bereichsangaben und Eigenschaften von Properties sinnvoll und umfassend?
        \item keine Inkonsistenzen!
    \end{itemize}
    \item Verwendung anspruchsvoller Konstrukte
    \begin{itemize}
        \item Komplexe Klassen
        \item GCIs
        \item Mehr als Datenbank / RDFS
    \end{itemize}
    \item interessante Folgerungen
    \begin{itemize}
        \item neue Unterklassenbeziehung
        \item neue Instanzen für Individuen
    \end{itemize}
    \item Präsentationsstil
    \begin{itemize}
        \item Einführung in Domäne
        \item Designprozess (entsprechend OD101-Leitfaden)
        \item Wesentliche Designentscheidungen
        \item Gefundene und entfernte Inkonsistenzen oder sonstige Fehler
        \item Live-Demo
    \end{itemize}
\end{itemize}

\subsection{Tips zur Modellierung}

Dos

\begin{itemize}
    \item Entitäten sinnvoll in Klassen und Individuen einteilen
    \begin{itemize}
        \item FootballSport ist ein Individuum, der Sport „American Football“ (und damit ggf. Instanz der Klassen Sport und Game)
        \item Football ist die Klasse aller Bälle, mit denen man FootballSport spielen kann.
    \end{itemize}
    \item Unterschied zwischen Instanz- und Unterklassenbeziehung beachten
    \begin{itemize}
        \item Lucy ist Instanz der Klasse Child
        \item Child ist Unterklasse von Human
    \end{itemize}
    \item Semantik von „Unterklasse“ beachten
    \begin{itemize}
        \item X SubClassOf Y bedeutet: Jedes X ist ein Y
        \item Bildschirm ist keine Unterklasse von Laptop, sondern ein teil
        \item Sensor ist keine Unterklasse von Messwert, er hat Messwerte
        \item Zeitintervall ist keine Unterklasse von Dauer, sie hat eine Dauer
    \end{itemize}
    \item bei hoher Anzahl von Geschwisterklassen über neue Klassenebene nachdenken
    \begin{itemize}
        \item Wenn man 50 Unterklassen von Tier hat, neue Klassen Säugetier, Fisch, Insekt einführen
    \end{itemize}
\end{itemize}

Don'ts

\begin{itemize}
    \item Von Einfachheit der Manchester-Syntax verführen lassen
    \begin{itemize}
        \item Nur weil ein Manchester-Ausdruck so ähnlich klingt wie ein englischer Satz, sagt er nicht notwendig dasselbe aus
    \end{itemize}
    \item Etwas falsches sagen, weil das, was man meint, in OWL nicht formalisierbar ist
    \begin{itemize}
        \item Besser nichts sagen, als etwas falsches sagen.
        \item ggf. eine allgemeinere Aussage machen.
    \end{itemize}
\end{itemize}

\subsection{Themenfindung}

    \begin{enumerate}
        \item Thema: Worum geht es?  
        
            Ausschnitt aus der Originaltrilogie (Episode IV-VI)  
        \item Quellen: Wo kann verlässliche Information zur Domäne gefunden werden? 
        
            Star Wars Wikis, \href{https://starwars.fandom.com/wiki/Star_Wars}{Fandom}, \href{https://www.jedipedia.net/wiki/Jedipedia:Hauptseite}{Jedipedia}, Filmhandlungen und Charakterbeschreibungen aus den Episoden 4--6
        \item Zweck: Wozu soll die Ontologie dienen? 
        
            Informationen zu Beziehungen von Charakteren, Spezies und Fraktionen liefern sowie daraus neue Schlussfolgerungen ableiten 
        \item Wer soll die Ontologie benutzen? 
        
        Fans
        \item Detaillierungsgrad: Was sind die allgemeinsten und speziellsten Begriffe? 
        
            Allgemein: Klassen für Lebewesen, Fraktionen, Orte und relevante Gegenstände 

            \item Speziell: Einzelne Figuren, Droiden, Schiffe und ikonische Artefakte 
        

        \item Blickwinkel: Welche Weltsicht liegt zugrunde? 
        
        Sicht des Lesers 

        Fiktive Welt wird real beschrieben 
        \item Kompetenzfragen: Welche Fragen sollen beantwortet werden? 
        
        \begin{itemize}
            \item Welche Figuren gehören zu den Rebellen, zum Imperium oder zu den Jedi?
            \item Welche Figuren sind miteinander verwandt, insbesondere innerhalb der Skywalker-Familie?
            \item Welche Spezies besitzen typische Eigenschaften, die für die Modellierung relevant sind?
            \item Welche Figuren nutzen oder besitzen bestimmte Droiden, Schiffe oder Lichtschwerter?
            \item Welche Schlussfolgerungen lassen sich aus Identität, Zugehörigkeit und Verwandtschaft ableiten?
        \end{itemize}

        \item Grenzen: Was wird nicht behandelt? 
        
        \begin{itemize}
            \item Nur die Originaltrilogie, keine Prequels, Sequels oder Expanded Universe Inhalte
            \item Keine vollständige Modellierung aller Orte, Planeten oder Schlachten
            \item Keine detaillierte Abbildung von Zeitverläufen oder jeder einzelnen Filmszene
            \item Keine behind-the-scenes Informationen zu Schauspielern, Produktion oder Veröffentlichung
        \end{itemize}

    \end{enumerate}

\subsubsection{Klassen Sammlung}

\begin{itemize}
    \item \textbf{Person} -- Oberklasse für alle relevanten Figuren
    \item \textbf{Charakter} -- konkrete handelnde Figur der Handlung
    \item \textbf{Mensch}, \textbf{Droid}, \textbf{Alien} -- Speziesklassen
    \item \textbf{Jedi}, \textbf{Sith}, \textbf{Rebel}, \textbf{Imperial} -- Fraktionen bzw. Rollen
    \item \textbf{FamilyMember} -- Figuren mit modellierter Verwandtschaft
    \item \textbf{Vehicle}, \textbf{Starship}, \textbf{Speeder} -- relevante Fortbewegungsmittel
    \item \textbf{Artifact}, \textbf{Lightsaber}, \textbf{Weapon} -- wichtige Gegenstände
    \item \textbf{Planet}, \textbf{Location} -- begrenzte Ortsklassen für das Referat
\end{itemize}

Mögliche Properties:

\begin{itemize}
    \item \textbf{hasSpecies} -- verbindet Charakter mit Spezies
    \item \textbf{memberOf} -- verbindet Charakter mit Fraktion oder Orden
    \item \textbf{hasParent}, \textbf{hasChild}, \textbf{hasSibling} -- Verwandtschaft
    \item \textbf{uses}, \textbf{owns}, \textbf{pilots} -- Besitz und Nutzung
    \item \textbf{locatedOn} -- Zuordnung zu Orten oder Planeten
\end{itemize}

\subsubsection{Mögliche Ableitung}

\begin{itemize}
    \item Wenn Luke Skywalker als Sohn von Darth Vader und Darth Vader als Anakin Skywalker modelliert wird, kann die Ontologie Luke und Leia als Kinder derselben Person erkennen.
    \item Aus einer Modellierung von \emph{memberOf} und geeigneten Klassen kann ein Charakter automatisch als Rebel oder Imperial eingeordnet werden.
    \item Wenn eine Figur ein Lichtschwert besitzt und zugleich der Klasse Jedi zugeordnet ist, lassen sich typische Jedi-Figuren als Unterklasse oder durch äquivalente Klassen erkennen.
    \item Durch definierte Verwandtschaftsregeln kann ein neuer Wissensstand entstehen, ohne dass jede Beziehung explizit eingetragen werden muss.
\end{itemize}
