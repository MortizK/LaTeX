\section*{Grundlegende Rechenregeln}
\begin{compactenum}
\item Erwartungswert: Linearität
\[E[aX+b]=aE[X]+b,\qquad E\Big[\sum_i X_i\Big]=\sum_i E[X_i].\]
\item Varianz / Kovarianz:
\[\operatorname{Var}(X)=E[X^2]-E[X]^2,\qquad \operatorname{Cov}(X,Y)=E[XY]-E[X]E[Y].\]
\[\operatorname{Var}(aX+b)=a^2\operatorname{Var}(X),\quad \operatorname{Var}(X+Y)=\operatorname{Var}(X)+\operatorname{Var}(Y)+2\operatorname{Cov}(X,Y).
\]
\item Unabhängigkeit: $\Cov(X,Y)=0\Rightarrow\Var(X+Y)=\Var(X)+\Var(Y)$.
\item Matrixform (kurz): Für Vektor $X$ und Matrix $A,b$:
\[\operatorname{Var}(AX+b)=A\operatorname{Var}(X)A^T,\qquad E[AX+b]=AE[X]+b.\]
\end{compactenum}

\section*{Urnenmodelle}
\begin{compactenum}
\item Mit Zurücklegen (multinom): Ziehe $n$ aus $k$ Typen mit Wahrscheinlichkeiten $p_i$.
\[P(N_1=n_1,\dots)=\frac{n!}{\prod n_i!}\prod p_i^{n_i}.
\]
\item Ohne Zurücklegen: Hypergeometrisch (ein Typ): Ziehe $n$ aus $N$ mit $K$ Erfolgen:
\[P(X=x)=\frac{\binom{K}{x}\binom{N-K}{n-x}}{\binom{N}{n}},\quad E[X]=n\frac{K}{N},\ Var(X)=n\frac{K}{N}\frac{N-K}{N}\frac{N-n}{N-1}.
\]
\item Multivariate Hypergeometrisch (counts $n_i$):
\[P(\{n_i\})=\frac{\prod_i\binom{K_i}{n_i}}{\binom{N}{n}},\quad \sum n_i=n.\]
\end{compactenum}

\section*{Wichtige Verteilungen (nach Häufigkeit)}
\small
\subsection*{Normalverteilung $\mathcal N(\mu,\sigma^2)$}
\begin{compactenum}
\item Kurz: Kontinuierlich, Gauss'sche Glocke.
\item Notation: $X\sim\mathcal N(\mu,\sigma^2)$.
\item PDF: \[f(x)=\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{(x-\mu)^2}{2\sigma^2}}.\]
\item CDF: $F(x)=\Phi\big(\frac{x-\mu}{\sigma}\big)$ (Standardnormalverteilungsfunktion $\Phi$).
\item $E[X]=\mu,\;\Var(X)=\sigma^2$.
\end{compactenum}

\subsection*{Binomialverteilung $\mathrm{Bin}(n,p)$}
\begin{compactenum}
\item Kurz: Anzahl Erfolge in $n$ Bernoulli-Versuchen.
\item PMF: \[P(X=k)=\binom{n}{k}p^k(1-p)^{n-k}.
\]
\item $E[X]=np,\;\Var(X)=np(1-p)$.
\end{compactenum}

\subsection*{Poissonverteilung $\mathrm{Pois}(\lambda)$}
\begin{compactenum}
\item Kurz: Limit von Binomial bei großem $n$, kleinem $p$ mit $np=\lambda$.
\item PMF: \[P(X=k)=e^{-\lambda}\frac{\lambda^k}{k!}.
\]
\item $E[X]=\Var(X)=\lambda$.
\end{compactenum}

\subsection*{Exponentialverteilung $\mathrm{Exp}(\lambda)$}
\begin{compactenum}
\item Kurz: Wartezeit bei Poisson-Prozess.
\item PDF: $f(x)=\lambda e^{-\lambda x},\;x\ge0$; CDF: $1-e^{-\lambda x}$.
\item $E[X]=1/\lambda,\;\Var(X)=1/\lambda^2$.
\end{compactenum}

\subsection*{Student-$t$ mit $\nu$ Freiheitsgraden}
\begin{compactenum}
\item Kurz: Stichprobenmittel mit unbekannter Varianz (Normalgrundlage).
\item Notation: $t_\nu$. Erwartungswert $0$ (für $\nu>1$), Var= $\nu/(\nu-2)$ für $\nu>2$.
\end{compactenum}

\subsection*{Chi-Quadrat $\chi^2_k$}
\begin{compactenum}
\item Kurz: Summe von $k$ unabhängigen $Z^2$ mit $Z\sim N(0,1)$.
\item $E= k,\;\Var=2k$.
\end{compactenum}

\subsection*{Uniform $U(a,b)$}
\[f(x)=\frac{1}{b-a},\;E=\frac{a+b}{2},\;\Var=\frac{(b-a)^2}{12}.
\]

\subsection*{Bernoulli $\mathrm{Ber}(p)$}
\[P(X=1)=p,\;P(X=0)=1-p,\;E=p,\;\Var=p(1-p).
\]

\section*{ML-Schätzer (häufige Modelle)}
\small
\begin{compactenum}
\item Normal $X_i\stackrel{iid}{\sim}\mathcal N(\mu,\sigma^2)$:
\[L(\mu,\sigma)=\prod\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{(x_i-\mu)^2}{2\sigma^2}}.\]
Maximieren: \[\hat\mu=\bar x,\qquad \hat\sigma^2=\frac{1}{n}\sum (x_i-\bar x)^2.\]
\item Exponential $f(x)=\lambda e^{-\lambda x}$:\ \[\hat\lambda=\frac{n}{\sum x_i}=1/\bar x.\]
\item Bernoulli/Binomial: \[\hat p=\frac{\sum x_i}{n}.\]
\item Poisson: \[\hat\lambda=\bar x.\]
\item Uniform $U(0,\theta)$ (MLE): \[\hat\theta=\max_i X_i\ (\text{bias korr.: }\frac{n+1}{n}\max X_i).\]
\end{compactenum}
\normalsize

\section*{Konfidenzintervalle — Kurzleitfaden}
\begin{compactenum}
\item Schritte: Modellannahme → geeignete Statistik wählen → Verteilung der Statistik bestimmen → Quantile → CI bilden.
\item Tabelle wählen:
\begin{compactenum}
\item $N(0,1)$ bei $\sigma$ bekannt oder großen $n$-Approx. für Mittelwert/Anteile.
\item $t$-Tabelle bei Mittelwert mit unbekannter $\sigma$.
\item $\chi^2$-Tabelle bei Varianz / Standardabweichung eines Normalmodells.
\end{compactenum}
\item Normalfall, $\sigma$ bekannt:
\[\bar X\pm z_{1-\alpha/2}\frac{\sigma}{\sqrt n}.
\]
\item Normalfall, $\sigma$ unbekannt:
\[\bar X\pm t_{n-1,1-\alpha/2}\frac{s}{\sqrt n},\quad s^2=\frac{1}{n-1}\sum (x_i-\bar x)^2.
\]
\item Varianz CI (Normal):
\[\left(\frac{(n-1)s^2}{\chi^2_{n-1;1-\alpha/2}},\ \frac{(n-1)s^2}{\chi^2_{n-1;\alpha/2}}\right).
\]
\item Anteil $p$: \[\hat p\pm z_{1-\alpha/2}\sqrt{\frac{\hat p(1-\hat p)}{n}}\] (bei kleinen $n$ exakte Clopper–Pearson).
\item Umkehrrechnung Intervall $\to$ Konfidenzniveau:
\begin{compactenum}
\item Zentraler z/t-Fall: falls $c$ der verwendete Quantilwert ist, dann
\[1-\alpha=\begin{cases}
2\Phi(c)-1,& z\text{-Fall}\\
2F_{t,\nu}(c)-1,& t\text{-Fall}
\end{cases}
\quad\Rightarrow\quad \alpha=1-(\text{Konfidenzniveau}).\]
\item Bei beidseitigem Intervall immer: $\alpha=2\cdot$ linke Restwahrscheinlichkeit, also $c=z_{1-\alpha/2}$ bzw. $t_{\nu;1-\alpha/2}$.
\item Wenn nur die Intervallgrenzen $[L,U]$ gegeben sind und das Intervall symmetrisch ist: Mittelpunkt $m=\frac{L+U}{2}$, Halbbreite $h=\frac{U-L}{2}$, dann $c=h/\mathrm{SE}$ und daraus $1-\alpha$ über die Tabelle/Funktion.
\end{compactenum}
\end{compactenum}

\section*{Signifikanztests — Ablauf + Formeln}
\begin{compactenum}
\item Allgemein: $H_0$ vs. $H_1$, Teststatistik $T$, Verteilung unter $H_0$, p-Wert oder kritischer Bereich, Entscheidung.
\item Merkschema für die Wahl: 1 Stichprobe $\to$ Gauß/t/Binomial/$\chi^2$; 2 Stichproben $\to$ Differenz-/Zweistichprobentest; Häufigkeiten in Tabellen $\to$ Anpassung/Kontingenz; Zusammenhang zweier metrischer Größen $\to$ Korrelationstest.
\item Einstrichproben-Gaußtest: Mittelwerttest bei normalverteilten Daten mit bekannter $\sigma$.
\[T=\frac{\bar X-\mu_0}{\sigma/\sqrt n}\sim N(0,1).\]
Voraussetzung: i.i.d., Normalmodell oder großes $n$; $\sigma$ bekannt.
\item Binomialtest: Anteil/Erfolgschance in $n$ Bernoulli-Versuchen, exakt.
\[X\sim\mathrm{Bin}(n,p_0),\quad T=X.\]
Voraussetzung: unabhängige Bernoulli-Versuche, nur 0/1-Ausgänge, kleines bis mittleres $n$ oder exakter Test gewünscht.
\item Einstrichproben-$t$-Test: Mittelwerttest bei unbekannter $\sigma$.
\[T=\frac{\bar X-\mu_0}{s/\sqrt n}\sim t_{n-1}.\]
Voraussetzung: i.i.d., annähernd normal; bei großem $n$ robust.
\item Approximativer Gaußtest: für Anteile oder Mittelwerte mit großer Stichprobe.
\[T\approx N(0,1),\quad \hat p=\frac Xn,\quad T=\frac{\hat p-p_0}{\sqrt{p_0(1-p_0)/n}}.\]
Voraussetzung: $np\geq5$ und $n(1-p)\geq5$
\item Differenztests: Vergleich zweier Parameter.
\[T_{\bar X_1-\bar X_2}=\frac{(\bar X_1-\bar X_2)-\Delta_0}{\mathrm{SE}},\quad T_{\hat p_1-\hat p_2}=\frac{(\hat p_1-\hat p_2)-\Delta_0}{\mathrm{SE}}.\]
Unabhängige Stichproben: zwei Gruppen; gepaart: vor/nachher mit Differenzen $D_i$ und dann 1-Stichproben-$t$ auf $D_i$.
\item Chi-Quadrat-Test für Varianz: Normalmodell, Test von $\sigma^2$.
\[T=\frac{(n-1)s^2}{\sigma_0^2}\sim\chi^2_{n-1}.\]
Voraussetzung: Normalverteilung der Grundgesamtheit.
\item Zweistichprobentests: Mittelwerte oder Anteile zweier Gruppen.
\[T_{\mu_1-\mu_2}=\frac{(\bar X_1-\bar X_2)-\Delta_0}{\sqrt{s_1^2/n_1+s_2^2/n_2}}\quad(\text{Welch}),\]
bei gleicher Varianz ggf. gepoolter $t$-Test. Für Anteile: analog mit gepoolter/ungepoolter Standardfehlerformel.
Voraussetzung: unabhängige Stichproben, gleiche Varianz nur bei gepooltem $t$-Test.
\item Chi-Quadrat-Anpassungstest: Verteilung einer kategorialen Variable gegen Sollverteilung.
\[\chi^2=\sum_i\frac{(O_i-E_i)^2}{E_i}\approx\chi^2_{k-1-m}.\]
Voraussetzung: erwartete Häufigkeiten pro Klasse ideal $\ge 5$, Kategorien disjunkt und vollständig.
\item Kontingenztest: Unabhängigkeit zweier kategorialer Merkmale in Kreuztabelle.
\[\chi^2=\sum_{ij}\frac{(O_{ij}-E_{ij})^2}{E_{ij}},\quad E_{ij}=\frac{R_iC_j}{n}.\]
Voraussetzung: erwartete Zellenbesetzungen möglichst $\ge 5$; Freiheitsgrade $(r-1)(c-1)$.
\item Korrelationstest: Test auf linearen Zusammenhang $\rho=0$.
\[T=\frac{r\sqrt{n-2}}{\sqrt{1-r^2}}\sim t_{n-2}.\]
Voraussetzung: bivariate Normalverteilung oder zumindest annähernd linearer Zusammenhang ohne starke Ausreißer.
\item Entscheidung: Signifikanzniveau $\alpha$ wählen, $T$ berechnen, p-Wert bestimmen, ablehnen falls $p<\alpha$.
\item Häufige Formulierungen für Hypothesen:
\begin{compactenum}
\item "mindestens", "höchstens", "nicht kleiner als" $\to$ meist in $H_0$ mit $=$ bzw. $\ge/\le$.
\item "größer", "mehr", "besser", "erhöht" $\to$ $H_1: >$.
\item "kleiner", "weniger", "gesunken" $\to$ $H_1: <$.
\item "anders", "unterschiedlich", "verändert" $\to$ $H_1: \neq$.
\item Praktisch: $H_0$ enthält fast immer die Gleichheit; $H_1$ ist die gesuchte Abweichung.
\end{compactenum}
\end{compactenum}
